cmake_minimum_required(VERSION 3.20)
project(rpcs3_ios_policy_tests LANGUAGES CXX)

get_filename_component(RPCS3_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
option(RPCS3_IOS_TEST_SANITIZERS "Enable address/undefined-behavior sanitizers" OFF)
add_executable(ios_policy_tests policy_tests.cpp)
add_executable(ios_audio_policy_tests audio_policy_tests.cpp)
foreach(test_target ios_policy_tests ios_audio_policy_tests)
target_compile_features(${test_target} PRIVATE cxx_std_20)
set_target_properties(${test_target} PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(${test_target} PRIVATE "${RPCS3_SOURCE_ROOT}/rpcs3")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(RPCS3_IOS_TEST_SANITIZERS)
        target_compile_options(${test_target} PRIVATE -fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
        target_link_options(${test_target} PRIVATE -fsanitize=address,undefined,float-cast-overflow)
    endif()
endif()
endforeach()
enable_testing()
add_test(NAME ios_policy_tests COMMAND ios_policy_tests)
add_test(NAME ios_audio_policy_tests COMMAND ios_audio_policy_tests)
find_package(Python3 3.8 COMPONENTS Interpreter REQUIRED)
add_test(NAME ios_python_tests COMMAND "${Python3_EXECUTABLE}" -m unittest discover
    -s "${CMAKE_CURRENT_LIST_DIR}" -p "test_*.py" -v)
